cmake_minimum_required(VERSION 3.20)
project(DMTGeometryCore LANGUAGES CXX)
add_library(dmt_geometry INTERFACE)
target_include_directories(dmt_geometry INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(dmt_geometry INTERFACE cxx_std_20)
add_executable(geometry_tests tests/test.cpp)
target_link_libraries(geometry_tests PRIVATE dmt_geometry)
enable_testing()
add_test(NAME geometry_invariants COMMAND geometry_tests)

add_executable(atlas_render src/atlas.cpp)
target_link_libraries(atlas_render PRIVATE dmt_geometry)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(atlas_render PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME atlas_smoke COMMAND atlas_render ${CMAKE_CURRENT_BINARY_DIR}/smoke.ppm 0.7 96)

add_executable(atlas_advanced src/advanced_render.cpp)
target_link_libraries(atlas_advanced PRIVATE dmt_geometry)
add_executable(advanced_tests tests/advanced_test.cpp)
target_link_libraries(advanced_tests PRIVATE dmt_geometry)
add_test(NAME advanced_invariants COMMAND advanced_tests)
add_test(NAME advanced_smoke COMMAND atlas_advanced ${CMAKE_CURRENT_BINARY_DIR}/advanced_smoke.ppm 96 0.5 0)
